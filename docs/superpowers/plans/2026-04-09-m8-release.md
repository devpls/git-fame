- [ ] **Step 1: Write the test file**

Create `tests/e2e/cli.test.ts`:

```ts
import { spawnSync } from 'node:child_process';
import { join } from 'node:path';
import { rmSync } from 'node:fs';
import { afterAll, beforeAll, describe, expect, it } from 'vitest';
import { buildRepo } from '../helpers/build-repo.js';

const CLI = join(import.meta.dirname, '../../dist/cli/bin.js');
const repoRoot = join(import.meta.dirname, '../..');

const run = (
  args: string[],
  cwd = repoRoot,
): { stdout: string; stderr: string; status: number | null } => {
  const result = spawnSync('node', [CLI, ...args], {
    cwd,
    encoding: 'utf8',
    timeout: 60_000,
  });
  return { stdout: result.stdout, stderr: result.stderr, status: result.status };
};

describe('CLI E2E', () => {
  beforeAll(() => {
    const build = spawnSync('npm', ['run', 'build'], { cwd: repoRoot, encoding: 'utf8' });
    if (build.status !== 0) {
      throw new Error(`Build failed: ${build.stderr}`);
    }
  });

  const createdRepos: string[] = [];
  afterAll(() => {
    while (createdRepos.length > 0) {
      const dir = createdRepos.pop();
      if (dir !== undefined) {
        rmSync(dir, { recursive: true, force: true });
      }
    }
  });

  it('--help exits 0 and lists all flags', () => {
    const { stdout, status } = run(['--help']);
    expect(status).toBe(0);
    expect(stdout).toContain('--format');
    expect(stdout).toContain('--include-globs');
    expect(stdout).toContain('--submodules');
    expect(stdout).toContain('--rev');
    expect(stdout).toContain('--sort');
  });

  it('--version exits 0 and prints the version', () => {
    const { stdout, status } = run(['--version']);
    expect(status).toBe(0);
    expect(stdout.trim()).toMatch(/^\d+\.\d+\.\d+$/);
  });

  it('exits 0 and prints a table for a valid repo', () => {
    const dir = buildRepo([
      { author: 'Alice <a@x>', date: '2024-01-01T00:00:00Z', files: { 'a.txt': 'hello\n' } },
    ]);
    createdRepos.push(dir);

    const { stdout, status } = run([dir]);
    expect(status).toBe(0);
    expect(stdout).toContain('linesAlive');
    expect(stdout).toContain('Alice');
  });

  it('--format json produces valid JSON', () => {
    const dir = buildRepo([
      { author: 'Alice <a@x>', date: '2024-01-01T00:00:00Z', files: { 'a.txt': 'hello\n' } },
    ]);
    createdRepos.push(dir);

    const { stdout, status } = run(['--format', 'json', dir]);
    expect(status).toBe(0);
    const parsed = JSON.parse(stdout);
    expect(parsed.meta.version).toBe('0.1.0');
    expect(parsed.authors).toBeInstanceOf(Array);
  });

  it('--format csv starts with a header row', () => {
    const dir = buildRepo([
      { author: 'Alice <a@x>', date: '2024-01-01T00:00:00Z', files: { 'a.txt': 'hello\n' } },
    ]);
    createdRepos.push(dir);

    const { stdout, status } = run(['--format', 'csv', dir]);
    expect(status).toBe(0);
    expect(stdout.split('\n')[0]).toContain('author,linesAlive');
  });

  it('--format markdown contains table separator', () => {
    const dir = buildRepo([
      { author: 'Alice <a@x>', date: '2024-01-01T00:00:00Z', files: { 'a.txt': 'hello\n' } },
    ]);
    createdRepos.push(dir);

    const { stdout, status } = run(['--format', 'markdown', dir]);
    expect(status).toBe(0);
    expect(stdout).toContain('| --- |');
  });

  it('exits 1 for a non-existent path', () => {
    const { status, stderr } = run(['/this/path/does/not/exist/ever']);
    expect(status).toBe(1);
    expect(stderr).toContain('not a git repository');
  });

  it('exits 0 with --sort and --limit', () => {
    const dir = buildRepo([
      { author: 'Alice <a@x>', date: '2024-01-01T00:00:00Z', files: { 'a.txt': 'hello\n' } },
    ]);
    createdRepos.push(dir);

    const { status } = run(['--sort', 'commits', '--limit', '1', dir]);
    expect(status).toBe(0);
  });

  it('exits 0 with --rev', () => {
    const dir = buildRepo([
      { author: 'Alice <a@x>', date: '2024-01-01T00:00:00Z', files: { 'a.txt': 'v1\n' } },
      { author: 'Alice <a@x>', date: '2024-01-02T00:00:00Z', files: { 'a.txt': 'v2\n' } },
    ]);
    createdRepos.push(dir);

    const { status, stdout } = run(['--format', 'json', '--rev', 'HEAD~1', dir]);
    expect(status).toBe(0);
    expect(() => JSON.parse(stdout)).not.toThrow();
  });
});
```

```ts
it('handles detached HEAD gracefully', async () => {
  const { spawnSync } = await import('node:child_process');
  const dir = buildRepo([
    { author: 'Alice <a@x>', date: '2024-01-01T00:00:00Z', files: { 'a.txt': 'hi\n' } },
  ]);
  createdRepos.push(dir);

  // Detach HEAD
  const sha = spawnSync('git', ['rev-parse', 'HEAD'], { cwd: dir, encoding: 'utf8' }).stdout.trim();
  spawnSync('git', ['checkout', sha], { cwd: dir });

  const report = await analyze({ path: dir });
  expect(report.repo.headSha).toBe(sha);
  expect(report.authors).toHaveLength(1);
});

it('returns empty report with ALL_FILES_FILTERED warning when all files are generated', async () => {
  const dir = buildRepo([
    {
      author: 'Alice <a@x>',
      date: '2024-01-01T00:00:00Z',
      files: { 'package-lock.json': '{ "lockfileVersion": 3 }\n' },
    },
  ]);
  createdRepos.push(dir);

  const report = await analyze({ path: dir });

  // package-lock.json is generated + excluded by default
  // Only file is filtered, so linesAlive should be 0
  const totalAlive = report.authors.reduce((sum, a) => sum + a.linesAlive, 0);
  expect(totalAlive).toBe(0);
});
```

````markdown
# node-fame

Fast, accurate git contribution stats — lines, commits, files per author.

Inspired by [git-fame](https://github.com/casperdcl/git-fame) (Python), rewritten in TypeScript for correctness and speed.

## Features

- **Two metrics**: lines alive in HEAD (blame) + lines added/deleted (log)
- **Smart defaults**: excludes generated files, ignores whitespace changes, follows renames
- **Fast**: parallel blame via streaming parsers, 7-10x faster than git-fame
- **Flexible output**: table, JSON, CSV, Markdown
- **Multi-repo**: submodules, recursive workspace analysis
- **Library + CLI**: use as `npx node-fame` or `import { analyze } from 'node-fame'`

## Quick start

```bash
npx node-fame .
npx node-fame /path/to/repo
npx node-fame --format json .
npx node-fame --include-globs '*.ts' '*.tsx' .
```
````

## Installation

```bash
npm install -g node-fame    # global CLI
npm install node-fame       # library dependency
```

## CLI Usage

```
node-fame [options] [path]

Arguments:
  path                           Repository path (default: current directory)

Options:
  -V, --version                  Output the version number
  --format <format>              Output format: table, json, csv, markdown (default: "table")
  --sort <column>                Sort by: linesAlive, linesAdded, linesDeleted, commits, files (default: "linesAlive")
  --limit <n>                    Show only top N authors
  --rev <ref>                    Analyze at a specific commit, tag, or branch
  --from <ref>                   Start of commit range (used with --to)
  --to <ref>                     End of commit range (used with --from)
  --since <date>                 Only count log entries after this date
  --until <date>                 Only count log entries before this date
  --include-whitespace           Count whitespace-only changes
  --include-binary               Include binary files
  --include-generated            Include generated/vendored files
  --exclude-minified             Exclude minified files
  --no-follow-renames            Do not follow renames in blame
  --no-mailmap                   Do not apply .mailmap
  --include-globs <patterns...>  Only analyze matching files
  --exclude-globs <patterns...>  Exclude matching files
  --submodules                   Walk into submodules
  --split-submodules             Separate reports per submodule
  --recursive                    Analyze all repos in subdirectories
  -h, --help                     Display help
```

## Library API

```ts
import { analyze, render } from 'node-fame';

const report = await analyze({
  path: '/path/to/repo',
  include: { generated: false },
  options: { followRenames: true, applyMailmap: true },
});

console.log(render(report, 'table'));
console.log(render(report, 'json', { sort: { by: 'commits' }, limit: 10 }));
```

## Multi-repo

```ts
import { analyzeMany } from 'node-fame';

// Recursive workspace scan
const reports = await analyzeMany({ path: '/workspace', recursive: true });

// Split submodule reports
const reports = await analyzeMany({ path: '.', submodules: true, splitSubmodules: true });
```

## How it works

1. **Discover** — list tracked files, filter binaries and generated files
2. **Log** — `git log --numstat` for additions/deletions per author
3. **Blame** — `git blame --line-porcelain -w -M -C` per file (parallel)
4. **Assemble** — merge into a typed `Report`

```markdown
# Performance Notes

Benchmarks from development dogfood sessions on macOS (Apple Silicon).

## node-fame vs git-fame

Repository: `/Users/mike/work/store` (3,281 tracked files, ~550k total lines, ~500 commits)

| Tool                              | Filter                                     | Files analyzed | Wall time | Files/sec |
| --------------------------------- | ------------------------------------------ | -------------- | --------- | --------- |
| git-fame (Python, Docker)         | `--incl='\.(ts\|tsx\|css)$' --excl='lock'` | 2,797          | 5m 23s    | ~8.7      |
| node-fame (M3, no filters)        | none                                       | 3,281          | 45s       | ~73       |
| node-fame (M4a, generated filter) | default (excl generated)                   | ~2,100         | 49s       | ~43       |

**Summary**: ~7x faster than git-fame wall-clock on equivalent workload. The difference comes from:

- Streaming porcelain parsing (no full-output buffering)
- Parallel blame via `p-limit(os.cpus().length)` (809% CPU utilisation observed)
- Native Node.js child_process.spawn vs Python subprocess

## Correctness comparison

| Metric                          | git-fame | node-fame | Delta       |
| ------------------------------- | -------- | --------- | ----------- |
| Total lines (TS/TSX/CSS filter) | 135,159  | 134,336   | -823 (0.6%) |

The small difference is attributable to node-fame's default `-w` (whitespace ignore) and `-M -C` (rename tracking) in blame, which git-fame does not apply by default.
```
